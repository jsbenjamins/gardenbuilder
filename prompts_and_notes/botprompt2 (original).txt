Requirements:
- Path should be just one extra "item" to choose besides the existing bushes and trees.
- Path build history/trajectory should be kept such that a path can be extended logically according to a tileset of tiles that consists of 16 available tiles where on either north, east,south or west side of the tile the path can be continued. Current path building direction should also be kept track of from the second path tile placement onwards.
- the first tile used for starting a path is always closedA. When clicking directly north of the  start tile, the start tile should change to change to openN and the new tile should be openS such that the two placed tiles connect nicely. If the next tile (i.e. third path tile) is also placed directly north of the current path the starting tile stays the same (openN), the second tile become openNS and the third tile will be an openS. If the forth tile is also north, the third tile should change to openNS from openS and the fourth will be openS etc. Likewise if the fourth tile would be placed directly east of the third tile, the third tile would become openSE and the forth would be openE.
- when there is  empty tile space in between two consecutively placed path tiles, it should start with a closedA starting tile again and if the gap is clicked filled logically.
- for now have the bot either continues building the path in the current direction when following and places a random non-path element when not following. The bot should, for now, not delete path tiles.
- there are 16 tiles of exactly a grid cell in width and length (32x32): closedA (starting tiles, all connections closed). openN (pathtile with an opening on the north side to connect to tile north of it), openS, openW, openE, openNS, openWE, openSE,openNE, openSW, openNW, openNSE, openNSW, openWES, openWEN and openA. They all have extension ".png" and are located under the folder 'assets/path/'
- use openNS.png as the image for the list of items on the left
- when clicking a cell at the edge of the whole grid: at the north and south edge, use openNS.png as path tile, on the west and east edge use openWE.png as path tile

    // function to check tile availability
    function canPlace(type, x, y) {
      const s = spriteSizes[type];

      // bounds check (multi-tile)
      if (x < 0 || y < 0 || x + s.w > GRID_SIZE || y + s.h > GRID_SIZE) {
        return false;
      }

      // overlap check
      for (const p of placedSprites) {
        const overlap =
          x < p.x + p.w &&
          x + s.w > p.x &&
          y < p.y + p.h &&
          y + s.h > p.y;

        if (overlap) return false;
      }

      return true;
    }

    // for BOT determine position scores when following
    function scorePosition(type, x, y, humanX, humanY) {
      let score = 0;

      const category = spriteSizes[type].c;

      // ---------- 1. SYMMETRY (strong weight)
      const centerX = Math.floor(GRID_SIZE / 2);
      const centerY = Math.floor(GRID_SIZE / 2);

      const dx = humanX - centerX;
      const dy = humanY - centerY;

      const sx = centerX - dx;
      const sy = centerY - dy;

      const distToSymmetry = Math.abs(x - sx) + Math.abs(y - sy);

      // score += Math.max(0, 20 - distToSymmetry * 5); // strong pull
      score += BOT_SYMMETRY_STRENGTH * Math.max(0, 20 - distToSymmetry * 5);


      // ---------- 2. CLUSTERING (same category neighbors)
      let neighbors = 0;

      for (const p of placedSprites) {
        if (spriteSizes[p.type].c !== category) continue;

        const dist = Math.abs(x - p.x) + Math.abs(y - p.y);
        if (dist <= 2) neighbors++;
      }

      score += neighbors * 4;


      // ---------- 3. DISTANCE TO HUMAN (soft preference)
      const distToHuman = Math.abs(x - humanX) + Math.abs(y - humanY);
      score += Math.max(0, 10 - distToHuman);


      return score;
    }

    // for BOT to find best scoring position when following
    function findBestPosition(type, originX, originY) {
      let best = null;
      let bestScore = -Infinity;

      // search area (local but wide enough)
      for (let dx = -4; dx <= 4; dx++) {
        for (let dy = -4; dy <= 4; dy++) {
          const x = originX + dx;
          const y = originY + dy;

          if (!canPlace(type, x, y)) continue;

          const score = scorePosition(type, x, y, originX, originY);

          if (score > bestScore) {
            bestScore = score;
            best = { x, y };
          }
        }
      }

      return best;
    }

    // ---------- MULTI-TILE PLACEMENT ----------
    function placeSprite(type, gridX, gridY, isBot = false) {
      // check whether gird cell is already occupied
      if (!canPlace(type, gridX, gridY)) return;

      const img = new Image();
      img.src = spriteImages[type];

      img.onload = () => {
        const wat = spriteSizes[type];
        const widthPx = wat.w * CELL_SIZE;
        const heightPx = wat.h * CELL_SIZE;

        const sprite = new Konva.Image({
          id: `cell-${gridX}-${gridY}`,
          x: gridX * CELL_SIZE,
          y: gridY * CELL_SIZE,
          width: widthPx,
          height: heightPx,
          image: img,
          draggable: false,
          zIndex: zCounter++
        });

        sprite.scale({ x: 0.1, y: 0.1 });
        itemLayer.add(sprite);

        sprite.to({
          scaleX: 1,
          scaleY: 1,
          duration: 0.15,
          easing: Konva.Easings.EaseOut
        });

        itemLayer.draw();

        // track placement
        placedSprites.push({
          type,
          x: gridX,
          y: gridY,
          w: wat.w,
          h: wat.h,
          sprite
        });

        // only track HUMAN actions
        if (!isBot) {
          lastHumanAction = { type, x: gridX, y: gridY };
          triggerBot();
        }
      };
    }

    function deleteSprite(gridX, gridY, isBot = false) {
      const index = placedSprites.findIndex(p => p.x === gridX && p.y === gridY);
      if (index === -1) return;

      const obj = placedSprites[index];

      obj.sprite.to({
        scaleX: 0.1,
        scaleY: 0.1,
        duration: 0.15,
        onFinish: () => {
          obj.sprite.destroy();
          zCounter = Math.max(1, zCounter - 1);
          itemLayer.draw();
        }
      });

      placedSprites.splice(index, 1);

      if (!isBot) {
        lastHumanAction = { delete: true, x: gridX, y: gridY };
        triggerBot();
      }
    }

    function triggerBot() {
      const delayOptions = [300, 450, 800, 1300];
      const delay = delayOptions[Math.floor(Math.random() * delayOptions.length)];

      setTimeout(() => runBot(), delay);
    }

    function runBot() {
      if (botBusy) return;
      if (!lastHumanAction || lastHumanAction.delete) return;

      botBusy = true;

      const follow = Math.random() < BOT_FOLLOW_CHANCE;

      if (follow) {
        botFollow();
      } else {
        botRandom();
      }
    }

    // newer botFollow that acts more intentional (clustering, symmetry, follow and fallback)
    function botFollow() {
      const { type, x, y } = lastHumanAction;
      const category = spriteSizes[type].c;

      const options = Object.keys(spriteSizes).filter(t => spriteSizes[t].c === category);
      const botType = options[Math.floor(Math.random() * options.length)];

      const best = findBestPosition(botType, x, y);

      if (best) {
        return moveBotCursorTo(best.x, best.y, 'place', botType, () => {
          placeSprite(botType, best.x, best.y, true);
        });
      }

      // fallback (rare)
      botRandom();
    }

    function botRandom() {
      const doDelete = Math.random() < 0.5;

      if (doDelete && placedSprites.length > 0) {
        moveBotCursorTo(lastHumanAction.x, lastHumanAction.y, 'delete', null, () => {
          deleteSprite(lastHumanAction.x, lastHumanAction.y, true);
        });
        return;
      }

      const categories = [...new Set(Object.values(spriteSizes).map(s => s.c))];
      const randomCategory = categories[Math.floor(Math.random() * categories.length)];

      const options = Object.keys(spriteSizes).filter(t => spriteSizes[t].c === randomCategory);

      for (let i = 0; i < 10; i++) {
        const type = options[Math.floor(Math.random() * options.length)];
        const x = Math.floor(Math.random() * GRID_SIZE);
        const y = Math.floor(Math.random() * GRID_SIZE);

        if (canPlace(type, x, y)) {
          return moveBotCursorTo(x, y, 'place', type, () => {
            placeSprite(type, x, y, true);
          });
        }
      }

      botBusy = false;
    }

    function moveBotCursorTo(gridX, gridY, mode, type, callback) {
      if (mode === 'place' && !botPlaceReady) {
        // fallback: skip animation but still place
        if (callback) callback();
        botBusy = false;
        return;
      }
      const baseX = gridX * CELL_SIZE;
      const baseY = gridY * CELL_SIZE;

      // hide both first
      botPlaceCursor.visible(false);
      botDeleteCursor.visible(false);

      let cursor;

      if (mode === 'place') {
        cursor = botPlaceCursor;
      } else {
        cursor = botDeleteCursor;
      }

      // always 1 tile size
      cursor.width(CELL_SIZE);
      cursor.height(CELL_SIZE);

      // ---------- CURRENT POSITION (already initialized to center)
      const startX = cursor.x();
      const startY = cursor.y();

      // ---------- MISALIGNMENT (small offset)
      const offsetX = (Math.random() - 0.5) * 6; // -3 to +3 px
      const offsetY = (Math.random() - 0.5) * 6;

      const targetX = baseX + offsetX;
      const targetY = baseY + offsetY;

      cursor.scale({ x: 1, y: 1 });
      cursor.visible(true);

      // ---------- PRE-MOVE HESITATION ----------
      const preDelay = [80, 120, 160][Math.floor(Math.random() * 3)];

      setTimeout(() => {
        cursor.visible(true);
        // ---------- MOVE (slightly imperfect)
        cursor.to({
          x: targetX,
          y: targetY,
          duration: 0.35,
          easing: Konva.Easings.EaseInOut,
          onFinish: () => {

            // ---------- MICRO CORRECTION (snap into place)
            cursor.to({
              x: baseX,
              y: baseY,
              duration: 0.08,
              onFinish: () => {

                // ---------- HOVER DELAY ----------
                const hoverDelay = [120, 180, 250][Math.floor(Math.random() * 3)];

                setTimeout(() => {

                  // ---------- CLICK ----------
                  cursor.to({
                    scaleX: 0.8,
                    scaleY: 0.8,
                    duration: 0.08,
                    onFinish: () => {
                      cursor.to({
                        scaleX: 1,
                        scaleY: 1,
                        duration: 0.08,
                        onFinish: () => {

                          cursor.visible(false);

                          if (callback) callback();

                          botBusy = false;
                        }
                      });
                    }
                  });

                }, hoverDelay);
              }
            });

          }
        });

      }, preDelay);
    }

    // Must set listening(true) and pointer-events must not be blocked
