    function getConnectedPathTiles(startX, startY) {
      const visited = new Set();
      const stack = [[startX, startY]];
      const result = [];

      while (stack.length) {
        const [x, y] = stack.pop();
        const k = key(x, y);

        if (visited.has(k)) continue;
        if (!pathTiles[k]) continue;

        visited.add(k);
        result.push([x, y]);

        const neighbors = [
          [x+1,y], [x-1,y],
          [x,y+1], [x,y-1]
        ];

        neighbors.forEach(([nx, ny]) => {
          if (!visited.has(key(nx, ny))) {
            stack.push([nx, ny]);
          }
        });
      }

      return result;

    function recomputeAllPaths() {

        const dirs = [
          { dx:0, dy:-1, d:"N" },
          { dx:1, dy:0, d:"E" },
          { dx:0, dy:1, d:"S" },
          { dx:-1, dy:0, d:"W" }

        dirs.forEach(({dx,dy,d}) => {
          if (pathTiles[key(x+dx, y+dy)]) {
            conn[d] = true;
          }
        });
      });

      // 3. redraw everything
      Object.keys(pathTiles).forEach(k => {
        const [x, y] = k.split('-').map(Number);
        drawPathTile(x, y);
      });

    function placePath(type, x, y, isBot = false) {
      // 1. block overlap
      const blocked = placedSprites.some(p =>
        x < p.x + p.w &&
        x + 1 > p.x &&
        y < p.y + p.h &&
        y + 1 > p.y
      );
      if (blocked) return;

      // 2. prevent squares
      if (createsSquare(x, y)) return;

      const k = key(x, y);

      // 3. create tile if needed
      if (!pathTiles[k]) {
        pathTiles[k] = {
          sprite: null,
          connections: { N:false, E:false, S:false, W:false }
        };
      }

      // 4. FULL GLOBAL RECOMPUTE
      recomputeAllPaths();

      // 5. bot trigger
      if (!isBot) {
        lastHumanAction = { type, x, y };
        triggerBot();
      }
    }

    // function to check general tile availability
    function canPlace(type, x, y) {
      console.log(type);
      const s = spriteSizes[type];

      // bounds check (multi-tile)
      if (x < 0 || y < 0 || x + s.w > GRID_SIZE || y + s.h > GRID_SIZE) {
        return false;
      }

      // overlap check
      for (const p of placedSprites) {
        const overlap =
          x < p.x + p.w &&
          x + s.w > p.x &&
          y < p.y + p.h &&
          y + s.h > p.y;

        if (!overlap) continue;

        // allow path to overwrite path
        if (type === 'path' && p.type === 'path') continue;
        return false;
      }

      return true;
    }

    // for BOT determine position scores when following
    function scorePosition(type, x, y, humanX, humanY) {
      let score = 0;

      const category = spriteSizes[type].c;

      // ---------- 1. SYMMETRY (strong weight)
      const centerX = Math.floor(GRID_SIZE / 2);
      const centerY = Math.floor(GRID_SIZE / 2);

      const dx = humanX - centerX;
      const dy = humanY - centerY;

      const sx = centerX - dx;
      const sy = centerY - dy;

      const distToSymmetry = Math.abs(x - sx) + Math.abs(y - sy);

      // score += Math.max(0, 20 - distToSymmetry * 5); // strong pull
      score += BOT_SYMMETRY_STRENGTH * Math.max(0, 20 - distToSymmetry * 5);


      // ---------- 2. CLUSTERING (same category neighbors)
      let neighbors = 0;

      for (const p of placedSprites) {
        if (spriteSizes[p.type].c !== category) continue;

        const dist = Math.abs(x - p.x) + Math.abs(y - p.y);
        if (dist <= 2) neighbors++;
      }

      score += neighbors * 4;


      // ---------- 3. DISTANCE TO HUMAN (soft preference)
      const distToHuman = Math.abs(x - humanX) + Math.abs(y - humanY);
      score += Math.max(0, 10 - distToHuman);


      return score;
    }

    // for BOT to find best scoring position when following
    function findBestPosition(type, originX, originY) {
      let best = null;
      let bestScore = -Infinity;

      // search area (local but wide enough)
      for (let dx = -4; dx <= 4; dx++) {
        for (let dy = -4; dy <= 4; dy++) {
          const x = originX + dx;
          const y = originY + dy;

          if (!canPlace(type, x, y)) continue;

          const score = scorePosition(type, x, y, originX, originY);

          if (score > bestScore) {
            bestScore = score;
            best = { x, y };
          }
        }
      }

      return best;
    }

      const options = Object.keys(spriteSizes).filter(t => spriteSizes[t].c === category);
      const botType = options[Math.floor(Math.random() * options.length)];

      const best = findBestPosition(botType, x, y);

      if (best) {
        return moveBotCursorTo(best.x, best.y, 'place', botType, () => {
          placeSprite(botType, best.x, best.y, true);
        });
      }

      // fallback (rare)
      botRandom();
    }

    function botFollowPath() {
      console.log("reached botFollowPath");
      if (!currentPathDirection || !lastPathTile) {
        botBusy = false;
        return;
      }

      const dir = currentPathDirection;

      const nx = lastPathTile.x + dir.dx;
      const ny = lastPathTile.y + dir.dy;

      // bounds
      if (nx < 0 || ny < 0 || nx >= GRID_SIZE || ny >= GRID_SIZE) {
        botBusy = false;
        return;
      }

      // don't overwrite trees/bushes
      const blocked = placedSprites.some(p =>
        nx < p.x + p.w &&
        nx + 1 > p.x &&
        ny < p.y + p.h &&
        ny + 1 > p.y
      );

      if (blocked) {
        botBusy = false;
        return;
      }

      moveBotCursorTo(nx, ny, 'place', 'path', () => {
        placePath('path',nx, ny, true);
      });
    }

    function botRandom() {
      const doDelete = Math.random() < 0.5;

      if (doDelete && placedSprites.length > 0) {
        moveBotCursorTo(lastHumanAction.x, lastHumanAction.y, 'delete', null, () => {
          deleteSprite(lastHumanAction.x, lastHumanAction.y, true);
        });
        return;
      }

      const categories = [...new Set(Object.values(spriteSizes).map(s => s.c))];
      const randomCategory = categories[Math.floor(Math.random() * categories.length)];

      //const options = Object.keys(spriteSizes).filter(t => spriteSizes[t].c === randomCategory);
      const options = Object.keys(spriteSizes).filter(t => spriteSizes[t].c === randomCategory && t !== 'path');

      for (let i = 0; i < 10; i++) {
        const type = options[Math.floor(Math.random() * options.length)];
        const x = Math.floor(Math.random() * GRID_SIZE);
        const y = Math.floor(Math.random() * GRID_SIZE);

        if (canPlace(type, x, y)) {
          return moveBotCursorTo(x, y, 'place', type, () => {
            placeSprite(type, x, y, true);
          });
        }
      }

      botBusy = false;
    }

    function moveBotCursorTo(gridX, gridY, mode, type, callback) {
      if (mode === 'place' && !botPlaceReady) {
        // fallback: skip animation but still place
        if (callback) callback();
        botBusy = false;
        return;
      }
      const baseX = gridX * CELL_SIZE;
      const baseY = gridY * CELL_SIZE;

      // hide both first
      botPlaceCursor.visible(false);
      botDeleteCursor.visible(false);

      let cursor;

      if (mode === 'place') {
        cursor = botPlaceCursor;
      } else {
        cursor = botDeleteCursor;
      }

      // always 1 tile size
      cursor.width(CELL_SIZE);
      cursor.height(CELL_SIZE);

      // ---------- CURRENT POSITION (already initialized to center)
      const startX = cursor.x();
      const startY = cursor.y();

      // ---------- MISALIGNMENT (small offset)
      const offsetX = (Math.random() - 0.5) * 6; // -3 to +3 px
      const offsetY = (Math.random() - 0.5) * 6;

      const targetX = baseX + offsetX;
      const targetY = baseY + offsetY;

      cursor.scale({ x: 1, y: 1 });
      cursor.visible(true);

      // ---------- PRE-MOVE HESITATION ----------
      const preDelay = [80, 120, 160][Math.floor(Math.random() * 3)];

      setTimeout(() => {
        cursor.visible(true);
        // ---------- MOVE (slightly imperfect)
        cursor.to({
          x: targetX,
          y: targetY,
          duration: 0.35,
          easing: Konva.Easings.EaseInOut,
          onFinish: () => {

            // ---------- MICRO CORRECTION (snap into place)
            cursor.to({
              x: baseX,
              y: baseY,
              duration: 0.08,
              onFinish: () => {

                // ---------- HOVER DELAY ----------
                const hoverDelay = [120, 180, 250][Math.floor(Math.random() * 3)];

                setTimeout(() => {

                  // ---------- CLICK ----------
                  cursor.to({
                    scaleX: 0.8,
                    scaleY: 0.8,
                    duration: 0.08,
                    onFinish: () => {
                      cursor.to({
                        scaleX: 1,
                        scaleY: 1,
                        duration: 0.08,
                        onFinish: () => {

                          cursor.visible(false);

                          if (callback) callback();

                          botBusy = false;
                        }
                      });
                    }
                  });

                }, hoverDelay);
              }
            });

          }
        });

      }, preDelay);
    }

    // ---------- FIXED KONVA CLICK DETECTION ----------
    // Must set listening(true) and pointer-events must not be blocked
    bgLayer.listening(true);
    itemLayer.listening(true);
    stage.listening(true);

    stage.on('pointerdown', () => {
      const pos = stage.getPointerPosition();
      if (!pos) return; // safety

      const gridX = Math.floor(pos.x / CELL_SIZE);
      const gridY = Math.floor(pos.y / CELL_SIZE);

      // Tool: erase
      if (selectedTool === 'erase') {
        deleteSprite(gridX, gridY);
        return;
      }

      // Tool: pick
      if (selectedTool === 'pick') {
        const existing = itemLayer.findOne(`#cell-${gridX}-${gridY}`);
        if (existing) {
          const src = existing.image().src;
          const type = Object.keys(spriteImages).find(t => spriteImages[t] === src);

          if (type) {
            clearSelectedUIitem();
            document.querySelector(`[data-type="${type}"]`).classList.add('selected');
            selectedType = type;
          }
        }
        return;
      }

      // Placement mode
      if (selectedType === 'path') {
        console.log("placing path");
        placePath(selectedType, gridX, gridY);
        return;
      }

      if (selectedType) {
        console.log("placing elements");
        deleteSprite(gridX, gridY);
        placeSprite(selectedType, gridX, gridY);
      }
    });

      } else {
        container.style.cursor = "url('assets/cursor-place-human.png') 16 16, auto";
      }
    }
